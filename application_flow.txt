async function api(path,opt={}){
  const r=await fetch(path,{method:opt.method||'GET',headers:{'Content-Type':'application/json'},body:opt.body?JSON.stringify(opt.body):undefined});
  if(r.status===401){location.href='/login';throw new Error('login')}
  const d=await r.json().catch(()=>({}));
  if(!r.ok)throw new Error(typeof d.detail==='string'?d.detail:'Something went wrong.');
  return d;
}

let stream=null;
async function openEnroll(){
  ['e_id','e_name','e_dep'].forEach(i=>$('#'+i).value='');$('#e_consent').checked=$('#e_replace').checked=false;msg($('#e_msg'),'','err');

# ---------- password + secret key ----------
def hash_password(pw: str, salt: Optional[str] = None) -> str:
    salt = salt or secrets.token_hex(8)
    h = hashlib.pbkdf2_hmac("sha256", pw.encode(), salt.encode(), 200_000).hex()
    return f"{salt}${h}"


def check_password(pw: str, stored: str) -> bool:
    try:
        salt, _ = stored.split("$", 1)
    except ValueError:
        return False
    return hmac.compare_digest(hash_password(pw, salt), stored)


def secret_key() -> str:
    p = BASE / "secret.key"
    if not p.exists():
        p.write_text(secrets.token_hex(32))
    return p.read_text().strip()


# ---------- in-memory gallery of enrolled faces ----------
class Gallery:
    def __init__(self):
        self.ids, self.names, self.embs = [], [], np.zeros((0, 512), np.float32)

    def reload(self, con):
        rows = con.execute(
            "SELECT emp_id, name, embedding FROM employees WHERE active = 1").fetchall()
        rows = [r for r in rows if r["embedding"] and len(r["embedding"]) == EMB_BYTES]
        self.ids = [r["emp_id"] for r in rows]
        self.names = [r["name"] for r in rows]
        self.embs = (np.array([np.frombuffer(r["embedding"], dtype=np.float32) for r in rows])
                     if rows else np.zeros((0, 512), np.float32))

    def best(self, emb, exclude=None):
        if len(self.ids) == 0:
            return None, 0.0
        sims = self.embs @ emb
        if exclude in self.ids:
            sims[self.ids.index(exclude)] = -1
        i = int(np.argmax(sims))
        return i, float(sims[i])

MESSAGES = {
    "no_face": "Look at the camera",
    "multiple": "One person at a time, please",
    "too_far": "Move closer to the camera",
}


LIVENESS_FRONT_YAW = 15       # must start roughly facing the camera
LIVENESS_TURN_YAW = 18        # degrees of head turn that proves a live person
LIVENESS_WINDOW = 15          # seconds allowed to complete the turn
LIVENESS_RELAX = 0.12         # a turned face matches a little lower, so relax the threshold at the end
STREAK_WINDOW = 8             # (liveness off) max seconds between two recognised frames
pending = {}                  # emp_id -> {"base": yaw, "start": time, "dev": max turn so far, "fails": n}


def _last_event(con, emp_id):
    today = datetime.now().strftime("%Y-%m-%d")
    return con.execute(
        "SELECT timestamp, event FROM attendance WHERE emp_id = ? AND date = ? AND voided = 0 "
        "ORDER BY timestamp DESC LIMIT 1", (emp_id, today)).fetchone()


def _already(name, last_ev, min_gap):
    """Return the 'already marked' reply if the last punch is too recent, else None."""
    if last_ev is None:
        return None
    last_t = datetime.strptime(last_ev["timestamp"], "%Y-%m-%d %H:%M:%S")
    if (datetime.now() - last_t).total_seconds() / 60 < min_gap:
        t = last_ev["timestamp"][11:16]
        return {"status": "already", "name": name, "event": last_ev["event"], "time": t,
                "message": f"{name}, already marked {last_ev['event']} at {t}"}
    return None


def _mark(emp_id, name, min_gap):
    with write_lock, dbm.db() as con:
        last_ev = _last_event(con, emp_id)
        dup = _already(name, last_ev, min_gap)
        if dup:
            return dup
        event = "IN" if last_ev is None or last_ev["event"] == "OUT" else "OUT"
        ts = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        con.execute("INSERT INTO attendance (emp_id, name, timestamp, date, event) VALUES (?,?,?,?,?)",
                    (emp_id, name, ts, ts[:10], event))
    streaks.pop(emp_id, None)
    return {"status": "marked", "name": name, "event": event, "time": ts[11:16],
            "message": f"{name}, checked {event} at {ts[11:16]}"}


@app.post("/api/recognize")
def recognize(body: FrameBody):
    img = engine.decode_image(body.image)
    if img is None:
        return {"status": "no_face", "message": MESSAGES["no_face"]}
    with dbm.db() as con:
        s = dbm.get_settings(con)
    thr, min_gap = float(s["threshold"]), int(s["min_gap_minutes"])
    use_liveness = s.get("liveness", "1") == "1" and engine.has_pose()
    now = time.time()
    for k in [k for k, p in pending.items() if now - p["start"] > LIVENESS_WINDOW]:
        pending.pop(k)

    # ---- stage 2: a head turn is in progress. Cheap frames (pose only), identity re-checked at the end.
    if use_liveness and pending:
        emp_id = min(pending, key=lambda k: pending[k]["start"])
        p = pending[emp_id]
        face, err = engine.single_face_light(img)
        if err:
            return {"status": err, "message": MESSAGES[err]}
        p["dev"] = max(p["dev"], abs(float(face.pose[1]) - p["base"]))
        if p["dev"] < LIVENESS_TURN_YAW:
            return {"status": "challenge", "message": "Slowly turn your head to one side"}
        engine.embed(img, face)
        i, score = gallery.best(face.normed_embedding)
        if i is None or gallery.ids[i] != emp_id or score < thr - LIVENESS_RELAX:
            p["fails"] += 1
            if p["fails"] >= 3:                       # a different person finished the turn
                pending.pop(emp_id)
                return {"status": "unknown", "message": "Face not recognised"}
            return {"status": "challenge", "message": "Now look back at the camera"}
        pending.pop(emp_id)
        return _mark(emp_id, gallery.names[i], min_gap)

    # ---- stage 1: identify the person
    t0 = time.time()
    face, err = engine.single_face(img)
    if time.time() - t0 > 2:
        print(f"Slow frame: {time.time() - t0:.1f}s (the PC is busy or the CPU is slow)")
    if err:
        return {"status": err, "message": MESSAGES[err]}
    i, score = gallery.best(face.normed_embedding)
    if i is None or score < thr:
        return {"status": "unknown", "message": "Face not recognised"}
    emp_id, name = gallery.ids[i], gallery.names[i]

    with dbm.db() as con:
        dup = _already(name, _last_event(con, emp_id), min_gap)
    if dup:
        return dup

    if use_liveness:
        yaw = float(face.pose[1])
        if abs(yaw) > LIVENESS_FRONT_YAW:
            return {"status": "face_front", "message": "Look straight at the camera"}
        pending[emp_id] = {"base": yaw, "start": now, "dev": 0.0, "fails": 0}
        return {"status": "challenge", "message": "Slowly turn your head to one side"}

    count, last = streaks.get(emp_id, (0, 0))
    count = count + 1 if now - last < STREAK_WINDOW else 1
    streaks[emp_id] = (count, now)
    if count < 2:
        return {"status": "verifying", "message": f"Verifying {name}..."}
    return _mark(emp_id, name, min_gap)

@app.post("/api/admin/employees", dependencies=[Depends(require_admin)])
def enroll(body: EnrollBody):
    emp_id, name = body.emp_id.strip(), body.name.strip()
    if not ID_RE.match(emp_id):
        raise HTTPException(400, "Employee ID may use letters, numbers, - and _ (max 20).")
    if not name:
        raise HTTPException(400, "Name is required.")
    if not body.consent:
        raise HTTPException(400, "The employee must consent to storing their face data.")
    if len(body.images) < 3:
        raise HTTPException(400, "At least 3 photos are needed.")

    embs, problems = [], set()
    for data in body.images[:8]:
        img = engine.decode_image(data)
        if img is None:
            continue
        face, err = engine.single_face(img, min_width=110)
        if err:
            problems.add(err)
        else:
            embs.append(face.normed_embedding)
    if len(embs) < 3:
        hint = ("Only one person should be in view and the face should fill more of the frame."
                if problems else "Photos could not be read.")
        raise HTTPException(400, f"Only {len(embs)} clear photos were captured. {hint}")

    emb = np.mean(embs, axis=0)
    emb = (emb / np.linalg.norm(emb)).astype(np.float32)

    with write_lock, dbm.db() as con:
        exists = con.execute("SELECT 1 FROM employees WHERE emp_id = ?", (emp_id,)).fetchone()
        if exists and not body.replace:
            raise HTTPException(409, "This ID already exists. Tick 'Replace existing' to re-enrol.")
        i, score = gallery.best(emb, exclude=emp_id)
        if i is not None and score >= DUPLICATE_SIM:
            raise HTTPException(409, f"This face is already enrolled as {gallery.names[i]} ({gallery.ids[i]}).")
        if exists:
            con.execute("UPDATE employees SET name=?, department=?, embedding=?, active=1 WHERE emp_id=?",
                        (name, body.department.strip(), emb.tobytes(), emp_id))
        else:
            con.execute("INSERT INTO employees (emp_id, name, department, embedding, active) VALUES (?,?,?,?,1)",
                        (emp_id, name, body.department.strip(), emb.tobytes()))
        gallery.reload(con)
    return {"ok": True, "samples": len(embs)}

def parse_range(start: Optional[str], end: Optional[str]):
    try:
        e = date.fromisoformat(end) if end else date.today()
        s = date.fromisoformat(start) if start else e
    except ValueError:
        raise HTTPException(400, "Dates must look like 2026-11-18.")
    if s > e:

"""Daily database backups. Runs inside the server, so no extra scheduler is needed."""
import shutil
import sqlite3
import threading
import time
from datetime import datetime
from pathlib import Path

import database as dbm

BACKUP_DIR = dbm.BASE_DIR / "backups"
KEEP = 30
PATTERN = "attendance_????-??-??_??????.db"


def check_folder(path: str):
    """Return an error message if the folder can't be used for backups, else None."""
    p = Path(path)
    if not p.is_dir():
        return "That folder does not exist."
    try:
        probe = p / ".write_test"
        probe.write_text("ok")
        probe.unlink()
    except OSError:
        return "Cannot write to that folder."
    return None


def _prune(folder: Path):
    files = sorted(folder.glob(PATTERN))
    for old in files[:-KEEP]:
        try:
            old.unlink()
        except OSError:
            pass


def make_backup(extra_dir: str = "") -> dict:
    """Consistent copy of the live database (safe while the server is running)."""
    BACKUP_DIR.mkdir(exist_ok=True)
    name = f"attendance_{datetime.now():%Y-%m-%d_%H%M%S}.db"
    dest = BACKUP_DIR / name
    src, dst = sqlite3.connect(dbm.DB_PATH), sqlite3.connect(dest)
    try:
        src.backup(dst)
    finally:
        dst.close()
        src.close()
    _prune(BACKUP_DIR)
    result = {"file": name, "size_kb": round(dest.stat().st_size / 1024), "copied_to": None, "copy_error": None}
    if extra_dir:
        try:
            shutil.copy2(dest, Path(extra_dir) / name)
            _prune(Path(extra_dir))
            result["copied_to"] = extra_dir
        except OSError as e:
            result["copy_error"] = f"Could not copy to {extra_dir}: {e.strerror or e}"
    return result


def list_backups(limit: int = 8):
    if not BACKUP_DIR.exists():
        return []
    out = []
    for f in sorted(BACKUP_DIR.glob(PATTERN), reverse=True)[:limit]:
        out.append({"name": f.name, "size_kb": round(f.stat().st_size / 1024),
                    "time": f.name[11:21] + " " + f.name[22:24] + ":" + f.name[24:26]})
    return out


def _loop():
    time.sleep(30)
    while True:
        try:
            today = datetime.now().strftime("%Y-%m-%d")
            if not any(BACKUP_DIR.glob(f"attendance_{today}_*.db")):
                with dbm.db() as con:
                    extra = dbm.get_settings(con).get("backup_folder", "")
                r = make_backup(extra)
                print(f"Backup created: {r['file']}" + (f" | {r['copy_error']}" if r["copy_error"] else ""))
        except Exception as e:                      # never let a backup problem stop the server
            print("Backup failed:", e)
        time.sleep(600)


def start_scheduler():
    threading.Thread(target=_loop, daemon=True, name="backup").start()

import base64
import threading

import cv2
import numpy as np

_app = None
_lock = threading.Lock()


def _get_app():
    global _app
    if _app is None:
        from insightface.app import FaceAnalysis
        # Only the three models we use (skips age/gender and 106-point landmarks) and a smaller
        # detection size: about a third faster per frame on a CPU, same recognition results.
        _app = FaceAnalysis(name="buffalo_l", providers=["CPUExecutionProvider"],
                            allowed_modules=["detection", "recognition", "landmark_3d_68"])
        _app.prepare(ctx_id=0, det_size=(320, 320))
    return _app


def has_pose():
    """True when the model pack provides head pose (needed for the liveness check)."""
    return _app is not None and "landmark_3d_68" in _app.models


def warm_up():
    with _lock:
        _get_app()


def decode_image(data_url: str):
    if len(data_url) > 3_000_000:
        return None
    if "," in data_url:
        data_url = data_url.split(",", 1)[1]
    try:
        raw = base64.b64decode(data_url)
    except Exception:
        return None
    return cv2.imdecode(np.frombuffer(raw, np.uint8), cv2.IMREAD_COLOR)


def analyze(img):
    with _lock:
        return _get_app().get(img)


def face_area(f):
    x1, y1, x2, y2 = f.bbox
    return float((x2 - x1) * (y2 - y1))


def _pick(faces, min_width):
    faces = [x for x in faces if float(x.det_score) >= 0.5]
    if not faces:
        return None, "no_face"
    faces.sort(key=face_area, reverse=True)
    if len(faces) > 1 and face_area(faces[1]) > 0.5 * face_area(faces[0]):
        return None, "multiple"
    best = faces[0]
    if (best.bbox[2] - best.bbox[0]) < min_width:
        return None, "too_far"
    return best, None


def single_face(img, min_width=90):
    """Detection + pose + identity vector. Returns (face, error_code); error_code is None when OK."""
    return _pick(analyze(img), min_width)


def single_face_light(img, min_width=90):
    """Detection + head pose only (no identity vector): about 3x cheaper, used during the head turn."""
    from insightface.app.common import Face
    with _lock:
        app = _get_app()
        bboxes, kpss = app.det_model.detect(img, max_num=0, metric="default")
        faces = []
        for i in range(bboxes.shape[0]):
            fc = Face(bbox=bboxes[i, 0:4], kps=kpss[i] if kpss is not None else None, det_score=bboxes[i, 4])
            app.models["landmark_3d_68"].get(img, fc)
            faces.append(fc)
    return _pick(faces, min_width)


def embed(img, face):
    """Add the identity vector to a face found by single_face_light."""
    with _lock:
        _get_app().models["recognition"].get(img, face)
    return face

import sqlite3
import time
from datetime import datetime

import cv2
import numpy as np
from insightface.app import FaceAnalysis

DB = "attendance.db"
THRESHOLD = 0.45        # similarity needed to accept a match (higher = stricter)
COOLDOWN_SECONDS = 60   # don't log the same person again within this time
CONFIRM_FRAMES = 3      # must be recognised in this many frames in a row


def init_db():
    con = sqlite3.connect(DB)
    con.execute("""
        CREATE TABLE IF NOT EXISTS attendance (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            emp_id TEXT NOT NULL,
            name TEXT NOT NULL,
            timestamp TEXT NOT NULL,
            date TEXT NOT NULL,
            event TEXT NOT NULL
        )
    """)
    con.commit()
    return con


def load_employees(con):
    rows = con.execute("SELECT emp_id, name, embedding FROM employees").fetchall()
    ids = [r[0] for r in rows]
    names = [r[1] for r in rows]
    embs = np.array([np.frombuffer(r[2], dtype=np.float32) for r in rows])
    return ids, names, embs


def log_attendance(con, emp_id, name):
    now = datetime.now()
    today = now.strftime("%Y-%m-%d")
    # First scan of the day = IN, next scans alternate OUT, IN, OUT...
    count = con.execute(
        "SELECT COUNT(*) FROM attendance WHERE emp_id = ? AND date = ?",
        (emp_id, today),
    ).fetchone()[0]
    event = "IN" if count % 2 == 0 else "OUT"
    con.execute(
        "INSERT INTO attendance (emp_id, name, timestamp, date, event) VALUES (?, ?, ?, ?, ?)",
        (emp_id, name, now.strftime("%Y-%m-%d %H:%M:%S"), today, event),
    )
    con.commit()
    return event, now.strftime("%H:%M:%S")


def main():
    con = init_db()
    ids, names, embs = load_employees(con)
    if len(ids) == 0:
        print("No employees enrolled. Run enroll.py first.")
        return
    print(f"Loaded {len(ids)} employee(s).")

    app = FaceAnalysis(name="buffalo_l", providers=["CPUExecutionProvider"])
    app.prepare(ctx_id=0, det_size=(640, 640))

    cap = cv2.VideoCapture(0)
    last_logged = {}     # emp_id -> time of last log
    streak = {}          # emp_id -> consecutive frames matched
    banner = ""
    banner_until = 0
    print("Recognition running. Press Q in the camera window to quit.")

    while True:
        ok, frame = cap.read()
        if not ok:
            print("Cannot read from camera")
            break

        faces = app.get(frame)
        seen_now = set()

        for f in faces:
            x1, y1, x2, y2 = f.bbox.astype(int)
            sims = embs @ f.normed_embedding      # cosine similarity to each employee
            best = int(np.argmax(sims))
            score = float(sims[best])

            if score >= THRESHOLD:
                emp_id, name = ids[best], names[best]
                color = (0, 255, 0)
                label = f"{name} ({score:.2f})"
                seen_now.add(emp_id)
                streak[emp_id] = streak.get(emp_id, 0) + 1

                now = time.time()
                if (streak[emp_id] >= CONFIRM_FRAMES
                        and now - last_logged.get(emp_id, 0) > COOLDOWN_SECONDS):
                    event, t = log_attendance(con, emp_id, name)
                    last_logged[emp_id] = now
                    banner = f"{name}: {event} at {t}"
                    banner_until = now + 4
                    print(banner)
            else:
                color = (0, 0, 255)
                label = f"Unknown ({score:.2f})"

            cv2.rectangle(frame, (x1, y1), (x2, y2), color, 2)
            cv2.putText(frame, label, (x1, max(y1 - 10, 20)),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.7, color, 2)

        # reset streaks for people not seen this frame
        for emp_id in list(streak):
            if emp_id not in seen_now:
                streak[emp_id] = 0

        if time.time() < banner_until:
            cv2.putText(frame, banner, (10, 35),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.9, (0, 255, 0), 2)

        cv2.imshow("Attendance", frame)
        if cv2.waitKey(1) & 0xFF == ord("q"):
            break

    cap.release()
    cv2.destroyAllWindows()


if __name__ == "__main__":
    main()

from datetime import date, datetime, timedelta

WEEKDAYS = ["Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday", "Sunday"]


def _minutes(hhmm):
    h, m = hhmm.split(":")
    return int(h) * 60 + int(m)


def _format_duration(seconds):
    """Convert seconds into HH:MM format."""
    total_minutes = round(seconds / 60)
    hours, minutes = divmod(total_minutes, 60)
    return f"{hours:02d}:{minutes:02d}"


def build_report(con, start: date, end: date, s: dict):
    start_min = _minutes(s["office_start"])
    limit_min = start_min + int(s["grace_minutes"])
    off_days = {d.strip() for d in s["weekly_off"].split(",") if d.strip()}
    today = date.today()
    now_min = datetime.now().hour * 60 + datetime.now().minute

    emps = con.execute(
        "SELECT emp_id, name, department, active, created_at FROM employees ORDER BY name").fetchall()

    events = con.execute(
        "SELECT emp_id, timestamp, date, event, source FROM attendance WHERE date BETWEEN ? AND ? AND voided = 0 ORDER BY timestamp",
        (start.isoformat(), end.isoformat())).fetchall()

    holidays = {h["day"] for h in con.execute(
        "SELECT day FROM holidays WHERE day BETWEEN ? AND ?", (start.isoformat(), end.isoformat()))}

    leaves = {(l["emp_id"], l["day"]): l for l in con.execute(
        "SELECT emp_id, day, type, note FROM leaves WHERE day BETWEEN ? AND ?", (start.isoformat(), end.isoformat()))}

    by = {}
    for e in events:
        by.setdefault((e["emp_id"], e["date"]), []).append(e)

    rows, summary = [], {}

    for e in emps:
        summary[e["emp_id"]] = {
            "emp_id": e["emp_id"],
            "name": e["name"],
            "department": e["department"] or "",
            "present": 0,
            "late": 0,
            "absent": 0,
            "leave": 0,
            "hours": 0
        }

    d = start
    while d <= end:
        dstr, wd = d.isoformat(), WEEKDAYS[d.weekday()]

        for e in emps:
            evs = by.get((e["emp_id"], dstr), [])

            base = {
                "date": dstr,
                "weekday": wd[:3],
                "emp_id": e["emp_id"],
                "name": e["name"],
                "department": e["department"] or ""
            }

            sm = summary[e["emp_id"]]

            if not evs:
                if not e["active"] or d > today or wd in off_days or dstr in holidays:
                    continue

                if dstr < (e["created_at"] or "")[:10]:
                    continue

                lv = leaves.get((e["emp_id"], dstr))

                if lv:
                    sm["leave"] += 1
                    rows.append({
                        **base,
                        "first_in": "",
                        "last_out": "",
                        "hours": "00:00",
                        "status": "Leave",
                        "note": lv["type"] + (f": {lv['note']}" if lv["note"] else ""),
                        "scans": 0
                    })
                    continue

                if d == today and now_min <= limit_min:
                    status = "Not in yet"
                else:
                    status = "Absent"
                    sm["absent"] += 1

                rows.append({
                    **base,
                    "first_in": "",
                    "last_out": "",
                    "hours": "00:00",
                    "status": status,
                    "note": "",
                    "scans": 0
                })
                continue

            # Calculate completed IN/OUT sessions in seconds
            total_seconds = 0
            open_t, last_out = None, ""

            for ev in evs:
                t = datetime.strptime(ev["timestamp"], "%Y-%m-%d %H:%M:%S")

                if ev["event"] == "IN":
                    open_t = open_t or t

                elif open_t:
                    total_seconds += (t - open_t).total_seconds()
                    open_t = None
                    last_out = ev["timestamp"][11:16]

            first_in = evs[0]["timestamp"]
            fm = int(first_in[11:13]) * 60 + int(first_in[14:16])
            late = fm > limit_min

            note = ""

            if open_t:
                note = "Still in" if d == today else "No OUT recorded"

            if any(ev["source"] == "manual" for ev in evs):
                note = (note + "; " if note else "") + "Includes manual entry"

            # Daily hours in HH:MM format
            hours = _format_duration(total_seconds)

            sm["present"] += 1
            sm["late"] += 1 if late else 0

            # Keep summary internally in seconds
            sm["hours"] += total_seconds

            rows.append({
                **base,
                "first_in": first_in[11:16],
                "last_out": last_out,
                "hours": hours,
                "status": "Late" if late else "Present",
                "note": note,
                "scans": len(evs),
                "late_by": fm - start_min if late else 0
            })

        d += timedelta(days=1)

    summary_rows = [
        v for v in summary.values()
        if v["present"] or v["absent"] or v["leave"]
    ]

    # Convert summary total seconds into HH:MM format
    for v in summary_rows:
        v["hours"] = _format_duration(v["hours"])

    return rows, summary_rows
